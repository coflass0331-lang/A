async function fetchWeather(){
const {lat,lon}=state.loc;
const url='https://api.open-meteo.com/v1/forecast'
+`?latitude=${lat}&longitude=${lon}`
+'&daily=weather_code,temperature_2m_max,temperature_2m_min,apparent_temperature_max,apparent_temperature_min,precipitation_probability_max,precipitation_sum,wind_speed_10m_max,uv_index_max'
+'&hourly=temperature_2m,apparent_temperature'
+'&wind_speed_unit=ms&timezone=auto&forecast_days=8';
const res=await fetch(url);
if(!res.ok) throw new Error('API error');
const j=await res.json();
const hr=(date,h)=>{
const k=j.hourly.time.indexOf(`${date}T${pad(h)}:00`);
return k<0?null:{t:j.hourly.temperature_2m[k],f:j.hourly.apparent_temperature[k]};
};
return j.daily.time.map((date,i)=>({
date, code:j.daily.weather_code[i],
tmax:j.daily.temperature_2m_max[i], tmin:j.daily.temperature_2m_min[i],
fmax:j.daily.apparent_temperature_max[i], fmin:j.daily.apparent_temperature_min[i],
pop:j.daily.precipitation_probability_max[i]??0,
rain:j.daily.precipitation_sum[i]??0,
wind:j.daily.wind_speed_10m_max[i]??0,
uv:j.daily.uv_index_max[i]??0,
h9:hr(date,9),h15:hr(date,15),h21:hr(date,21)
}));
}

function recommend(day){
const base = day.fmax*0.6 + day.fmin*0.4; // 낮 시간 비중을 높인 체감온도
const eff = base + state.sens; // 체질 보정
const band = pickBand(eff);
const range = day.tmax - day.tmin;
const w = WMO(day.code);
const extras=[]; const reasons=[];
reasons.push({i:'🌡️',t:`체감온도 약 <b>${r(base)}°C</b>${state.sens?` (체질 보정 ${state.sens>0?'+':''}${state.sens}° 반영 → <b>${r(eff)}°C</b>)`:''} 기준으로 <b>'${band.name}'</b> 구간에 해당해요. ${band.desc}`});
if(range>=10){
extras.push(['🧥','벗고 입기 쉬운 겉옷']);
reasons.push({i:'↕️',t:`하루 기온 차이가 <b>${r(range)}°C</b>로 커요. 아침·저녁엔 쌀쌀하고 낮엔 더우니 <b>겹쳐 입고 낮에 벗을 수 있게</b> 준비하세요.`});
}
const snow = w.k==='snow';
if(snow){
extras.push(['👢','미끄럼 방지 신발'],['🧤','장갑'],['☂️','우산']);
reasons.push({i:'❄️',t:'눈 예보가 있어요. 길이 미끄러울 수 있으니 <b>바닥이 미끄럽지 않은 신발</b>과 방한 소품을 챙기세요.'});
} else if(day.pop>=50 || day.rain>=3 || w.k==='rain'){
extras.push(['☂️','우산'],['🥾','방수 신발']);
reasons.push({i:'☔',t:`강수확률 <b>${r(day.pop)}%</b>${day.rain>0?`, 예상 강수량 ${day.rain.toFixed(1)}mm`:''}예요. <b>우산</b>을 챙기고 물에 강한 신발·마르기 쉬운 소재를 추천해요.`});
} else if(day.pop>=30){
extras.push(['☂️','접이식 우산']);
reasons.push({i:'🌂',t:`강수확률이 <b>${r(day.pop)}%</b>라 갑작스러운 비에 대비해 <b>접이식 우산</b>이 있으면 안심이에요.`});
}
if(day.wind>=7){
extras.push(['🧣','바람 막는 목 보호']);
if(eff>=12) extras.push(['🧥','바람막이']);
reasons.push({i:'💨',t:`최대 풍속이 <b>${day.wind.toFixed(1)}m/s</b>로 바람이 강해요. 체감온도가 더 낮아지니 <b>바람을 막아주는 겉옷</b>을 추천해요.`});
}
if(day.uv>=6 && eff>=15 && !snow){
extras.push(['🧴','선크림'],['🕶️','선글라스']);
reasons.push({i:'🌞',t:`자외선 지수가 <b>${r(day.uv)}</b>로 높아요. <b>선크림·선글라스·모자</b>로 피부를 보호하세요.`});
}
if(eff>=28){
reasons.push({i:'💧',t:'더위로 땀이 많이 나는 날이에요. 밝은 색, 통풍이 잘 되는 소재를 고르고 <b>수분 보충</b>도 잊지 마세요.'});
}
if(eff<0){
reasons.push({i:'🥶',t:'영하의 체감온도예요. 목·손·귀 등 <b>노출 부위를 감싸고</b> 가능하면 이동 시간을 줄이세요.'});
}
return {base,eff,band,w,extras,reasons,range};
}
